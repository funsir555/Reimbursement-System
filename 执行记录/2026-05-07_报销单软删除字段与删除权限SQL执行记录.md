```sql
WHERE TABLE_SCHEMA = 'finex_db'
  AND TABLE_NAME = 'pm_document_instance'
  AND COLUMN_NAME IN ('deleted', 'deleted_at', 'deleted_by');

SELECT INDEX_NAME, COLUMN_NAME
FROM information_schema.STATISTICS
WHERE TABLE_SCHEMA = 'finex_db'
  AND TABLE_NAME = 'pm_document_instance'
  AND INDEX_NAME = 'idx_pm_document_instance_deleted';

SELECT permission_code, permission_name, parent_id, sort_order, status
FROM sys_permission
WHERE permission_code = 'expense:documents:delete';

SELECT COUNT(*) AS super_admin_permission_count
FROM sys_role_permission rp
JOIN sys_role r ON r.id = rp.role_id
JOIN sys_permission p ON p.id = rp.permission_id
WHERE r.role_code = 'SUPER_ADMIN'
  AND p.permission_code = 'expense:documents:delete';
```
